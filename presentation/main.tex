% 4
\begin{frame}{Problème métier}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{itemize}
        \item Les équipes assurance consultent des règles, documents et historiques dispersés.
        \item Les réponses doivent être rapides mais aussi justifiables.
        \item Les décisions sensibles nécessitent une validation humaine.
      \end{itemize}
    \column{0.42\textwidth}
      \begin{alertblock}{Risque si l'IA est isolée}
        Une réponse fluide mais non sourcée peut être inutilisable dans un contexte contractuel ou réglementaire.
      \end{alertblock}
  \end{columns}
  \note{Timing: 1 minute. Motiver le besoin de spécialisation, de sources et de garde-fous.}
\end{frame}

% 5
\begin{frame}{Objectifs du projet}
  \begin{itemize}
    \item Fournir une API et une interface simples pour interagir avec des agents métier.
    \item Séparer clairement frontend, backend, orchestration, outils métier et recherche documentaire.
    \item Rendre chaque réponse plus traçable grâce aux sources, aux contrôles et à l'historique.
    \item Conserver un repli déterministe si le fournisseur LLM est indisponible.
  \end{itemize}
  \note{Timing: 45 secondes. Insister sur le côté prototype robuste: le système doit continuer à répondre même sans LLM.}
\end{frame}

\begin{frame}{Flux d'une requête}
  \begin{enumerate}
    \item L'utilisateur envoie une question depuis le frontend.
    \item L'API authentifie la requête et transmet message + contexte.
    \item A0 route vers l'agent métier le plus adapté.
    \item Les outils métier exécutent les contrôles et récupèrent les sources.
    \item Le LLM synthétise si le provider est disponible.
    \item La réponse et la trace sont persistées dans l'historique.
  \end{enumerate}
  \note{Timing: 1 minute. Faire le lien avec une question concrète: délai de déclaration d'un sinistre.}
\end{frame}

% 10
\begin{frame}{RAG documentaire avec Qdrant}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{itemize}
        \item Extraction du texte des PDF.
        \item Découpage en fragments chevauchants.
        \item Embedding multilingue MiniLM.
        \item Stockage dans \texttt{insurance\_documents}.
        \item Recherche + reranking selon la question.
      \end{itemize}
    \column{0.44\textwidth}
      \begin{exampleblock}{Bénéfice}
        Les réponses peuvent citer des passages, pages et documents, au lieu de s'appuyer uniquement sur la mémoire du modèle.
      \end{exampleblock}
  \end{columns}
  \note{Timing: 1 minute. Expliquer que Qdrant stocke vecteur + payload documentaire.}
\end{frame}
