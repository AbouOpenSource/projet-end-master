\begin{frame}{Synthèse et allocation sous contrôle}

  \begin{block}{Question de recherche}
    Comment utiliser les sorties d'un LLM dans une chaîne multifactorielle tout en pouvant vérifier ses chiffres, ses sources et ses décisions ?
  \end{block}
  \begin{itemize}\raggedright
    \item \textbf{Synthèse :} restituer un état quantitatif déjà calculé.
    \item \textbf{Allocation :} proposer des poids soumis aux contrôles Python.
    \item \textbf{Évaluation :} comparer les notes, puis le PnL simulé.
  \end{itemize}
  \takeaway{Le moteur déterministe conserve le calcul et le contrôle des positions.}
  \source{Mémoire, introduction et architecture.}
  \oral{45 s. Dans la première expérience, le LLM ne modifie pas le portefeuille. Dans la seconde, une proposition acceptée change les positions simulées. Aucun ordre réel n'est envoyé et aucune approbation humaine effective n'est enregistrée.}{Mémoire, introduction et architecture.}
\end{frame}

\begin{frame}{Une expérience historique à portée limitée}

  \begin{itemize}\raggedright
    \item \textbf{Préentraînement :} appels en septembre 2026 sur des décisions datées de 2021--2025.
    \item \textbf{Données :} cours ajustés disponibles aujourd'hui, sans versions historiques point-in-time.
    \item \textbf{Échantillon :} quatre ETF et une seule trajectoire de 56 réponses.
    \item \textbf{Simulation :} clôture suivante, frais fixes, sans modèle de spread ni d'impact.
  \end{itemize}
  \takeaway{Le résultat technique est vérifiable. La généralisation financière reste à tester.}
  \source{Mémoire, sections 5.6.5 et 6.2.}
